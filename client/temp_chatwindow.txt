import React, { useState, useContext, useEffect, useRef } from 'react';
import { useTheme } from '../context/useTheme';
import { ChatContext } from '../context/ChatContext';
import { Send, Mic, Square, Play, Pause, X, Smile, Paperclip, PhoneCall, PhoneOff, MicOff } from 'lucide-react';
import { API_BASE } from '../api';
import RoomDetails from './RoomDetails';
import './ChatWindow.css';

const ChatWindow = ({ socket }) => {
  const theme = useTheme();
  const { currentRoom, messages, setMessages, currentUser, currentDM, typingRoom, typingDM, onlineUsers, loadingMessages, setSelectedProfileUser, blockedUsers, setBlockedUsers } = useContext(ChatContext);
  const [messageInput, setMessageInput] = useState('');
  const messagesContainerRef = useRef(null);
  const typingTimeoutRef = useRef(null);
  const loadingIntervalRef = useRef(null);
  const loadingFinishTimeoutRef = useRef(null);
  const [hoveredMessageId, setHoveredMessageId] = useState(null);
  const [uploadingAttachment, setUploadingAttachment] = useState(null);
  const [selectedFile, setSelectedFile] = useState(null);
  const [callStatus, setCallStatus] = useState('idle'); // 'idle' | 'calling' | 'ringing' | 'in-call'
  const [callPartnerId, setCallPartnerId] = useState(null);
  const [incomingOffer, setIncomingOffer] = useState(null);
  const localAudioRef = useRef(null);
  const remoteAudioRef = useRef(null);
  const pcRef = useRef(null);
  const localStreamRef = useRef(null);
  const [isMuted, setIsMuted] = useState(false);
  const [callNote, setCallNote] = useState('');
  const [messageMenuId, setMessageMenuId] = useState(null);
  const [dmPartnerLastSeen, setDmPartnerLastSeen] = useState(null);
  const [loadingProgress, setLoadingProgress] = useState(0);
  const [showRoomDetails, setShowRoomDetails] = useState(false);
  const [isRecording, setIsRecording] = useState(false);
  const [recordingTime, setRecordingTime] = useState(0);
  const [audioBlob, setAudioBlob] = useState(null);
  const mediaRecorderRef = useRef(null);
  const audioChunksRef = useRef([]);
  const recordingIntervalRef = useRef(null);
  const [showEmojiPicker, setShowEmojiPicker] = useState(false);
  const inputRef = useRef(null);
  const fileInputRef = useRef(null);
  const loadingPercent = Math.max(0, Math.min(100, Math.round(loadingProgress)));

  const scrollToEnd = (smooth = true) => {
    try {
      const el = messagesContainerRef.current;
      if (el) {
        el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' });
      }
    } catch {}
  };

  useEffect(() => {
    scrollToEnd(true);
  }, [messages]);

  // Jump to bottom when entering a DM or room
  useEffect(() => {
    setTimeout(() => scrollToEnd(false), 0);
  }, [currentDM, currentRoom]);

  // Fetch blocked users on mount
  useEffect(() => {
    if (!currentUser?.id) return;
    fetch(`${API_BASE}/api/friends/blocked/${currentUser.id}`)
      .then(res => res.json())
      .then(data => {
        setBlockedUsers(Array.isArray(data) ? data.map(u => u._id) : []);
      })
      .catch(() => {});
  }, [currentUser?.id, setBlockedUsers]);

  // Recording timer
  useEffect(() => {
    if (isRecording) {
      recordingIntervalRef.current = setInterval(() => {
        setRecordingTime(prev => prev + 1);
      }, 1000);
    } else {
      if (recordingIntervalRef.current) {
        clearInterval(recordingIntervalRef.current);
      }
      setRecordingTime(0);
    }
    return () => {
      if (recordingIntervalRef.current) {
        clearInterval(recordingIntervalRef.current);
      }
    };
  }, [isRecording]);

  // After loading finishes and loader disappears, ensure bottom
  useEffect(() => {
    if (!loadingMessages && loadingProgress === 0) {
      setTimeout(() => scrollToEnd(false), 0);
    }
  }, [loadingMessages, loadingProgress]);

  // Reset menu when switching context
  useEffect(() => {
    setMessageMenuId(null);
    setHoveredMessageId(null);
  }, [currentDM, currentRoom]);

  // Smooth pseudo progress for message loading
  useEffect(() => {
    if (!loadingMessages) return undefined;

    if (loadingFinishTimeoutRef.current) {
      clearTimeout(loadingFinishTimeoutRef.current);
      loadingFinishTimeoutRef.current = null;
    }
    setLoadingProgress(5);

    loadingIntervalRef.current = setInterval(() => {
      setLoadingProgress(prev => {
        if (prev >= 92) return prev;
        const increment = 4 + Math.random() * 6;
        return Math.min(prev + increment, 92);
      });
    }, 160);

    return () => {
      if (loadingIntervalRef.current) {
        clearInterval(loadingIntervalRef.current);
        loadingIntervalRef.current = null;
      }
    };
  }, [loadingMessages]);

  useEffect(() => {
    if (loadingMessages) return undefined;

    if (loadingIntervalRef.current) {
      clearInterval(loadingIntervalRef.current);
      loadingIntervalRef.current = null;
    }
    if (loadingProgress === 0) return undefined;

    setLoadingProgress(100);
    loadingFinishTimeoutRef.current = setTimeout(() => {
      setLoadingProgress(0);
      loadingFinishTimeoutRef.current = null;
    }, 320);

    return () => {
      if (loadingFinishTimeoutRef.current) {
        clearTimeout(loadingFinishTimeoutRef.current);
        loadingFinishTimeoutRef.current = null;
      }
    };
  }, [loadingMessages, loadingProgress]);

  useEffect(() => () => {
    if (loadingIntervalRef.current) clearInterval(loadingIntervalRef.current);
    if (loadingFinishTimeoutRef.current) clearTimeout(loadingFinishTimeoutRef.current);
  }, []);

  // Fetch blocked users list on mount
  useEffect(() => {
    if (!currentUser?.id) return;
    fetch(`${API_BASE}/api/friends/blocked/${currentUser.id}`)
      .then(res => res.json())
      .then(data => {
        setBlockedUsers(Array.isArray(data) ? data.map(u => u._id) : []);
      })
      .catch(() => {});
  }, [currentUser?.id]);

  // Fetch DM partner's last seen (only if offline)
  useEffect(() => {
    if (!currentDM?.userId) {
      setDmPartnerLastSeen(null);
      return;
    }
    // Check if user is online
    const isOnline = Array.isArray(onlineUsers) && onlineUsers.some(u => u.userId === currentDM.userId);
    if (isOnline) {
      setDmPartnerLastSeen(null);
      return;
    }
    // Only fetch last seen if offline
    fetch(`${API_BASE}/api/users/${currentDM.userId}`)
      .then(res => res.json())
      .then(data => {
        if (data.lastSeen) setDmPartnerLastSeen(data.lastSeen);
      })
      .catch(() => {});
  }, [currentDM, onlineUsers]);

  // Close menu on click outside
  useEffect(() => {
    const handleClickOutside = () => setMessageMenuId(null);
    if (messageMenuId) {
      document.addEventListener('click', handleClickOutside);
      return () => document.removeEventListener('click', handleClickOutside);
    }
  }, [messageMenuId]);

  const handleSendMessage = async (e) => {
    e.preventDefault();
    if (!messageInput.trim() || !currentUser) return;

    // DM
    if (currentDM) {
      const tempId = 'tmp_' + Date.now() + '_' + Math.random().toString(36).slice(2, 9);
      const optimistic = {
        tempId,
        sender: { _id: currentUser.id, username: currentUser.username, avatar: currentUser.avatar },
        senderName: currentUser.username,
        receiver: currentDM.userId,
        content: messageInput,
        timestamp: new Date()
      };
      setMessages(prev => [...prev, optimistic]);
      scrollToEnd();
      const contentToSend = messageInput;
      setMessageInput('');

      try {
        if (socket && socket.connected) {
          socket.emit('send_dm', {
            senderId: currentUser.id,
            senderName: currentUser.username,
            receiverId: currentDM.userId,
            content: contentToSend,
            tempId
          });
        }
        const res = await fetch(`${API_BASE}/api/dm/send`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ senderId: currentUser.id, receiverId: currentDM.userId, content: contentToSend })
        });
        const data = await res.json();
        if (!res.ok && (data?.error === 'blocked' || res.status === 403)) {
          setMessages(prev => prev.map(m => (
            m.tempId === tempId
              ? {
                  tempId,
                  sender: { _id: currentUser.id, username: currentUser.username },
                  senderName: currentUser.username,
                  receiver: currentDM.userId,
                  content: 'Message failed to send',
                  status: 'error',
                  timestamp: new Date()
                }
              : m
          )));
        } else {
          setMessages(prev => prev.map(m => (m.tempId === tempId ? data : m)));
        }
        scrollToEnd();
      } catch (err) {
        console.error('Error sending DM:', err);
      }
      return;
    }

    // Room
    if (socket && currentRoom) {
      const tempId = 'tmp_' + Date.now() + '_' + Math.random().toString(36).slice(2, 9);
      const optimistic = {
        tempId,
        sender: currentUser.id,
        senderName: currentUser.username,
        room: currentRoom,
        content: messageInput,
        timestamp: new Date()
      };
      setMessages(prev => [...prev, optimistic]);
      socket.emit('send_message', {
        content: messageInput,
        room: currentRoom,
        sender: currentUser.id,
        senderName: currentUser.username,
        tempId
      });
      setMessageInput('');
      scrollToEnd();
    }
  };

  // Emit typing (debounced)
  const emitTyping = (isTyping) => {
    if (!socket || !currentUser) return;
    if (currentDM) {
      socket.emit('typing_dm', { toUserId: currentDM.userId, fromUserId: currentUser.id, username: currentUser.username, isTyping });
    } else if (currentRoom) {
      socket.emit('typing', { room: currentRoom, userId: currentUser.id, username: currentUser.username, isTyping });
    }
  };

  const handleInputChange = (e) => {
    const val = e.target.value;
    setMessageInput(val);
    emitTyping(true);
    if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
    typingTimeoutRef.current = setTimeout(() => emitTyping(false), 1000);
  };

  const insertEmoji = (emoji) => {
    try {
      const input = inputRef.current;
      const start = input ? input.selectionStart : messageInput.length;
      const end = input ? input.selectionEnd : start;
      const next = (messageInput || '').slice(0, start) + emoji + (messageInput || '').slice(end);
      setMessageInput(next);
      setShowEmojiPicker(false);
      setTimeout(() => {
        if (input) {
          input.focus();
          const pos = start + emoji.length;
          input.setSelectionRange(pos, pos);
        }
      }, 0);
    } catch {}
  };

  // WebRTC helpers
  const createPeerConnection = () => {
    const pc = new RTCPeerConnection({
      iceServers: [{ urls: 'stun:stun.l.google.com:19302' }]
    });
    pc.onicecandidate = (event) => {
      if (event.candidate && socket && currentDM) {
        socket.emit('rtc_ice_candidate', { toUserId: currentDM.userId, fromUserId: currentUser.id, candidate: event.candidate });
      }
    };
    pc.ontrack = (event) => {
      const [stream] = event.streams;
      if (remoteAudioRef.current) {
        remoteAudioRef.current.srcObject = stream;
        // Try to autoplay remote audio when available
        const playPromise = remoteAudioRef.current.play();
        if (playPromise && typeof playPromise.then === 'function') {
          playPromise.catch(() => {/* ignore autoplay rejection */});
        }
      }
    };
    pcRef.current = pc;
    return pc;
  };

  const getLocalAudio = async () => {
    // Hint: Some browsers require an explicit permission prompt message
    setCallNote('Requesting microphone permissionâ€¦');
    if (navigator.permissions && navigator.permissions.query) {
      try {
        const status = await navigator.permissions.query({ name: 'microphone' });
        console.log('[RTC] mic permission status:', status.state);
        if (status.state === 'denied') {
          setCallNote('Microphone permission denied. Enable it in browser settings.');
        } else if (status.state === 'prompt') {
          setCallNote('Please allow microphone to start the call.');
        } else {
          setCallNote('Microphone granted. Connectingâ€¦');
        }
      } catch {}
    }
    const stream = await navigator.mediaDevices.getUserMedia({ audio: { echoCancellation: true, noiseSuppression: true } });
    localStreamRef.current = stream;
    if (localAudioRef.current) localAudioRef.current.srcObject = stream;
    return stream;
  };

  const startCall = async () => {
    if (!socket || !currentDM || !currentUser) return;
    try {
      setCallPartnerId(currentDM.userId);
      setCallStatus('calling');
      setCallNote('Callingâ€¦');
      const pc = createPeerConnection();
      const localStream = await getLocalAudio();
      if (!localStream || localStream.getAudioTracks().length === 0) {
        setCallNote('No microphone detected.');
      }
      localStream.getTracks().forEach(track => pc.addTrack(track, localStream));
      const offer = await pc.createOffer();
      await pc.setLocalDescription(offer);
      socket.emit('rtc_call_request', { toUserId: currentDM.userId, fromUserId: currentUser.id, offer });
    } catch (err) {
      console.error('startCall error', err);
      setCallNote('Failed to start call: ' + (err?.message || 'unknown error'));
      setCallStatus('idle');
    }
  };

  const acceptCall = async (offer, fromUserId) => {
    if (!socket || !currentUser) return;
    try {
      setCallPartnerId(fromUserId);
      setCallStatus('in-call');
      setCallNote('Connectingâ€¦');
      const pc = createPeerConnection();
      const localStream = await getLocalAudio();
      if (!localStream || localStream.getAudioTracks().length === 0) {
        setCallNote('No microphone detected.');
      }
      localStream.getTracks().forEach(track => pc.addTrack(track, localStream));
      await pc.setRemoteDescription(new RTCSessionDescription(offer));
      const answer = await pc.createAnswer();
      await pc.setLocalDescription(answer);
      socket.emit('rtc_call_answer', { toUserId: fromUserId, fromUserId: currentUser.id, answer });
      setIncomingOffer(null);
    } catch (err) {
      console.error('acceptCall error', err);
      setCallNote('Failed to accept call: ' + (err?.message || 'unknown error'));
      setCallStatus('idle');
    }
  };

  const endCall = () => {
    try {
      if (socket && callPartnerId) {
        socket.emit('rtc_call_end', { toUserId: callPartnerId, fromUserId: currentUser.id });
      }
    } catch {}
    setCallStatus('idle');
    setCallPartnerId(null);
    setCallNote('');
    if (pcRef.current) {
      try { pcRef.current.close(); } catch {}
      pcRef.current = null;
    }
    if (localStreamRef.current) {
      try { localStreamRef.current.getTracks().forEach(t => t.stop()); } catch {}
      localStreamRef.current = null;
    }
    if (remoteAudioRef.current) {
      remoteAudioRef.current.srcObject = null;
    }
  };

  const toggleMute = () => {
    setIsMuted(prev => {
      const next = !prev;
      const stream = localStreamRef.current;
      if (stream) stream.getAudioTracks().forEach(t => t.enabled = !next);
      return next;
    });
  };

  // Socket listeners for RTC
  useEffect(() => {
    if (!socket) return;
    const onIncomingCall = async ({ fromUserId, offer }) => {
      // Only ring if this is the active DM partner
      if (!currentDM || currentDM.userId !== fromUserId) return;
      setCallStatus('ringing');
      setIncomingOffer({ offer, fromUserId });
    };
    const onCallAnswer = async ({ fromUserId, answer }) => {
      try {
        if (pcRef.current) {
          await pcRef.current.setRemoteDescription(new RTCSessionDescription(answer));
          setCallStatus('in-call');
          setCallNote('Connected.');
        }
      } catch (err) {
        console.error('onCallAnswer error', err);
        setCallNote('Answer failed: ' + (err?.message || 'unknown error'));
      }
    };
    const onIceCandidate = async ({ fromUserId, candidate }) => {
      try {
        if (pcRef.current) {
          await pcRef.current.addIceCandidate(new RTCIceCandidate(candidate));
        }
      } catch (err) {
        console.error('addIceCandidate error', err);
      }
    };
    const onCallEnd = () => endCall();

    socket.on('rtc_incoming_call', onIncomingCall);
    socket.on('rtc_call_answer', onCallAnswer);
    socket.on('rtc_ice_candidate', onIceCandidate);
    socket.on('rtc_call_end', onCallEnd);
    return () => {
      socket.off('rtc_incoming_call', onIncomingCall);
      socket.off('rtc_call_answer', onCallAnswer);
      socket.off('rtc_ice_candidate', onIceCandidate);
      socket.off('rtc_call_end', onCallEnd);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [socket, currentDM?.userId]);

  const onAttachClick = () => {
    try {
      fileInputRef.current?.click();
    } catch {}
  };

  const onFileSelected = async (e) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const maxSizeMB = 10;
    if (file.size > maxSizeMB * 1024 * 1024) {
      alert(`File too large. Maximum size is ${maxSizeMB}MB`);
      e.target.value = '';
      return;
    }
    const mime = file.type || 'application/octet-stream';
    const name = file.name;
    const size = file.size;
    let kind = 'file';
    if (mime.startsWith('image/')) kind = 'image';
    else if (mime.startsWith('video/')) kind = 'video';
    
    // Read file for preview
    const reader = new FileReader();
    reader.onloadend = () => {
      setSelectedFile({
        base64: reader.result,
        mime,
        name,
        size,
        kind
      });
      e.target.value = '';
    };
    reader.readAsDataURL(file);
  };

  const sendSelectedFile = async () => {
    if (!selectedFile) return;
    const { base64, mime, name, size, kind } = selectedFile;
    setSelectedFile(null);
    await sendAttachmentData(base64, mime, name, size, kind);
  };

  const cancelSelectedFile = () => {
    setSelectedFile(null);
  };

  const sendAttachmentData = async (base64, mime, name, size, kind) => {
    let success = false;
    setUploadingAttachment({ name, size, kind, status: 'sending' });
    // DM
    if (currentDM) {
      const tempId = 'tmp_' + Date.now() + '_' + Math.random().toString(36).slice(2, 9);
      const optimistic = {
        tempId,
        sender: { _id: currentUser.id, username: currentUser.username, avatar: currentUser.avatar },
        senderName: currentUser.username,
        receiver: currentDM.userId,
        content: base64,
        type: kind,
        fileName: name,
        mimeType: mime,
        fileSize: size,
        timestamp: new Date()
      };
      setMessages(prev => [...prev, optimistic]);
      scrollToEnd();
      try {
        const payload = JSON.stringify({ senderId: currentUser.id, receiverId: currentDM.userId, content: base64, type: kind, fileName: name, mimeType: mime, fileSize: size, tempId });
        console.log('[sendAttachment] Payload size:', Math.round(payload.length / 1024), 'KB');
        
        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), 30000); // 30s timeout
        
        const res = await fetch(`${API_BASE}/api/dm/send`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: payload,
          signal: controller.signal
        });
        clearTimeout(timeoutId);
        if (!res.ok) {
          const errorText = await res.text();
          console.error('[sendAttachmentData] Server error:', res.status, errorText);
          throw new Error(`Server returned ${res.status}`);
        }
        const data = await res.json();
        if (data?.error === 'blocked') {
          setMessages(prev => prev.map(m => (
            m.tempId === tempId
              ? {
                  tempId,
                  sender: { _id: currentUser.id, username: currentUser.username },
                  senderName: currentUser.username,
                  receiver: currentDM.userId,
                  content: 'Message failed to send',
                  status: 'error',
                  timestamp: new Date()
                }
              : m
          )));
        } else {
          setMessages(prev => prev.map(m => (m.tempId === tempId ? data : m)));
          success = true;
        }
        scrollToEnd();
      } catch (err) {
        console.error('Error sending attachment DM:', err);
        if (err.name === 'AbortError') {
          alert('Upload timed out. File may be too large or connection is slow.');
        } else if (err.message.includes('413')) {
          alert('File too large for server. Try a smaller file.');
        }
      }
      if (success) {
        setUploadingAttachment({ name, size, kind, status: 'sent' });
        setTimeout(() => setUploadingAttachment(null), 1200);
      } else {
        setUploadingAttachment({ name, size, kind, status: 'error' });
      }
      return success;
    }

    // Room
    if (socket && currentRoom) {
      const tempId = 'tmp_' + Date.now() + '_' + Math.random().toString(36).slice(2, 9);
      const optimistic = {
        tempId,
        sender: currentUser.id,
        senderName: currentUser.username,
        room: currentRoom,
        content: base64,
        type: kind,
        fileName: name,
        mimeType: mime,
        fileSize: size,
        timestamp: new Date()
      };
      setMessages(prev => [...prev, optimistic]);
      socket.emit('send_message', {
        content: base64,
        type: kind,
        fileName: name,
        mimeType: mime,
        fileSize: size,
        room: currentRoom,
        sender: currentUser.id,
        senderName: currentUser.username,
        tempId
      });
      scrollToEnd();
      success = true;
    }
    if (success) {
      setUploadingAttachment({ name, size, kind, status: 'sent' });
      setTimeout(() => setUploadingAttachment(null), 1200);
    } else {
      setUploadingAttachment({ name, size, kind, status: 'error' });
    }
    return success;
  };

  const startRecording = async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
      const mediaRecorder = new MediaRecorder(stream);
      mediaRecorderRef.current = mediaRecorder;
      audioChunksRef.current = [];

      mediaRecorder.ondataavailable = (event) => {
        if (event.data.size > 0) {
          audioChunksRef.current.push(event.data);
        }
      };

      mediaRecorder.onstop = () => {
        const audioBlob = new Blob(audioChunksRef.current, { type: 'audio/webm' });
        setAudioBlob(audioBlob);
        stream.getTracks().forEach(track => track.stop());
      };

      mediaRecorder.start();
      setIsRecording(true);
    } catch (err) {
      console.error('Error accessing microphone:', err);
      alert('Could not access microphone. Please grant permission.');
    }
  };

  const stopRecording = () => {
    if (mediaRecorderRef.current && isRecording) {
      mediaRecorderRef.current.stop();
      setIsRecording(false);
    }
  };

  const sendRecordingDirectly = () => {
    if (mediaRecorderRef.current && isRecording) {
      // Stop recording and immediately send
      mediaRecorderRef.current.onstop = () => {
        const audioBlob = new Blob(audioChunksRef.current, { type: 'audio/webm' });
        mediaRecorderRef.current.stream.getTracks().forEach(track => track.stop());
        setIsRecording(false);
        
        // Send immediately
        const reader = new FileReader();
        reader.readAsDataURL(audioBlob);
        reader.onloadend = async () => {
          const base64Audio = reader.result;
          const duration = recordingTime;
          await sendAudioMessageData(base64Audio, duration);
        };
      };
      mediaRecorderRef.current.stop();
    }
  };

  const cancelRecording = () => {
    if (mediaRecorderRef.current && isRecording) {
      mediaRecorderRef.current.stop();
      setIsRecording(false);
    }
    setAudioBlob(null);
    audioChunksRef.current = [];
  };

  const sendAudioMessage = async () => {
    if (!audioBlob || !currentUser) return;

    const reader = new FileReader();
    reader.readAsDataURL(audioBlob);
    reader.onloadend = async () => {
      const base64Audio = reader.result;
      const duration = recordingTime;
      await sendAudioMessageData(base64Audio, duration);
      setAudioBlob(null);
    };
  };

  const sendAudioMessageData = async (base64Audio, duration) => {

      // DM
      if (currentDM) {
        const tempId = 'tmp_' + Date.now() + '_' + Math.random().toString(36).slice(2, 9);
        const optimistic = {
          tempId,
          sender: { _id: currentUser.id, username: currentUser.username, avatar: currentUser.avatar },
          senderName: currentUser.username,
          receiver: currentDM.userId,
          content: base64Audio,
          type: 'audio',
          duration,
          timestamp: new Date()
        };
        setMessages(prev => [...prev, optimistic]);
        scrollToEnd();
        setAudioBlob(null);

        try {
          if (socket && socket.connected) {
            socket.emit('send_dm', {
              senderId: currentUser.id,
              senderName: currentUser.username,
              receiverId: currentDM.userId,
              content: base64Audio,
              type: 'audio',
              duration,

            title="Attach file"
          >

          </div>
        </div>
      )}

      {/* In-Call Controls */}
      {callStatus === 'in-call' && (
        <div style={{ position: 'fixed', bottom: '1rem', right: '1rem', background: theme.colors.surface, color: theme.colors.text, border: `1px solid ${theme.colors.border}`, borderRadius: '0.75rem', padding: '0.75rem', display: 'flex', alignItems: 'center', gap: '0.75rem', zIndex: 9999 }}>

          {callNote && <span style={{ fontSize: '0.8rem', color: theme.colors.textMuted }}>{callNote}</span>}
          <button onClick={toggleMute} title={isMuted ? 'Unmute' : 'Mute'} style={{ padding: '0.5rem', borderRadius: '0.5rem', background: theme.colors.surface, border: `1px solid ${theme.colors.border}`, cursor: 'pointer', color: theme.colors.text }}>
            {isMuted ? <MicOff size={18} /> : <Mic size={18} />}
          </button>
          <button onClick={endCall} title="End call" style={{ padding: '0.5rem', borderRadius: '0.5rem', background: '#ef4444', color: '#fff', border: 'none', cursor: 'pointer' }}>
            <PhoneOff size={18} />
          </button>
        </div>
      )}
      
      {showRoomDetails && currentRoom && (
        <RoomDetails
          room={{ name: currentRoom }}
          onClose={() => setShowRoomDetails(false)}
        />
      )}
    </div>
  );
};

// Audio Player Component
const AudioPlayer = ({ audioData, duration, theme }) => {
  const [isPlaying, setIsPlaying] = useState(false);
  const [currentTime, setCurrentTime] = useState(0);
  const audioRef = useRef(null);

  useEffect(() => {
    if (!audioRef.current) return;
    const audio = audioRef.current;

    const handleTimeUpdate = () => setCurrentTime(audio.currentTime);
    const handleEnded = () => setIsPlaying(false);

    audio.addEventListener('timeupdate', handleTimeUpdate);
    audio.addEventListener('ended', handleEnded);

    return () => {
      audio.removeEventListener('timeupdate', handleTimeUpdate);
      audio.removeEventListener('ended', handleEnded);
    };
  }, []);

  const togglePlay = () => {
    if (!audioRef.current) return;
    if (isPlaying) {
      audioRef.current.pause();
    } else {
      audioRef.current.play();
    }
    setIsPlaying(!isPlaying);
  };

  const formatTime = (seconds) => {
    const mins = Math.floor(seconds / 60);
    const secs = Math.floor(seconds % 60);
    return `${mins}:${String(secs).padStart(2, '0')}`;
  };

  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: '0.75rem', minWidth: '200px' }}>
      <audio ref={audioRef} src={audioData} />
      <button
        onClick={togglePlay}
        style={{
          width: '36px',
          height: '36px',
          borderRadius: '50%',
          border: 'none',
          backgroundColor: 'rgba(255,255,255,0.2)',
          color: 'inherit',
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          cursor: 'pointer',
          transition: 'all 0.2s ease'
        }}
      >
        {isPlaying ? <Pause size={16} /> : <Play size={16} />}
      </button>
      <div style={{ flex: 1, display: 'flex', flexDirection: 'column', gap: '0.25rem' }}>
        <div style={{ width: '100%', height: '4px', borderRadius: '2px', backgroundColor: 'rgba(255,255,255,0.2)', overflow: 'hidden' }}>
          <div
            style={{
              width: `${audioRef.current ? (currentTime / audioRef.current.duration) * 100 : 0}%`,
              height: '100%',
              backgroundColor: 'rgba(255,255,255,0.8)',
              transition: 'width 0.1s linear'
            }}
          />
        </div>
        <div style={{ fontSize: '0.7rem', opacity: 0.8 }}>
          {audioRef.current ? formatTime(currentTime) : '0:00'} / {formatTime(duration || 0)}
        </div>
      </div>
    </div>
  );
};
